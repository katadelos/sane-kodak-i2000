cmake_minimum_required(VERSION 3.16)
project(sane_kodak_i2000 LANGUAGES C)
include(GNUInstallDirs)
include(CTest)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SANE REQUIRED IMPORTED_TARGET sane-backends)
pkg_check_modules(USB REQUIRED IMPORTED_TARGET libusb-1.0)
find_package(Threads REQUIRED)
set(base backend/kodak_i2000)
include(cmake/SaneSupport.cmake)
set(image_sources ${base}_image.c ${base}_color.c ${base}_enhancement.c)
set(capture_tables ${base}_capture_data.c ${base}_profile_piranha.c
    ${base}_profile_dt.c ${base}_camera_tables.c)
add_library(sane-kodak_i2000 MODULE ${base}.c ${base}_options.c ${base}_device_options.c
    ${base}_transport.c ${base}_boot.c ${base}_camera.c ${base}_scan.c
    ${base}_calibration.c ${base}_capture.c ${image_sources} ${base}_protocol.c
    ${capture_tables} ${base}_firmware.c $<TARGET_OBJECTS:kodak_support>)
target_include_directories(sane-kodak_i2000 PRIVATE "${sane_include}")
target_link_libraries(sane-kodak_i2000 PRIVATE PkgConfig::SANE PkgConfig::USB Threads::Threads m)
if(APPLE)
  set(plugin_suffix ".1.so")
else()
  set(plugin_suffix ".so.1")
  target_link_options(sane-kodak_i2000 PRIVATE -Wl,-z,defs)
endif()
set_target_properties(sane-kodak_i2000 PROPERTIES SUFFIX "${plugin_suffix}")
add_executable(kodak-i2000-fw tools/kodak-i2000-fw.c ${base}_firmware.c)
target_include_directories(kodak-i2000-fw PRIVATE .)
if(BUILD_TESTING)
  foreach(kind firmware image protocol)
    set(test kodak_i2000_${kind}_test)
    add_executable(${test} testsuite/backend/${test}.c ${base}_${kind}.c)
    target_include_directories(${test} PRIVATE . "${sane_include}")
    add_test(NAME ${test} COMMAND ${test})
  endforeach()
  target_sources(kodak_i2000_protocol_test PRIVATE ${capture_tables})
  target_sources(kodak_i2000_image_test PRIVATE ${base}_color.c ${base}_enhancement.c)
endif()
install(TARGETS sane-kodak_i2000 LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/sane)
install(TARGETS kodak-i2000-fw RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES backend/kodak_i2000.conf.in DESTINATION ${CMAKE_INSTALL_SYSCONFDIR}/sane.d RENAME kodak_i2000.conf)
install(FILES kodak_i2000 DESTINATION ${CMAKE_INSTALL_SYSCONFDIR}/sane.d/dll.d)
install(FILES doc/kodak-i2000-fw.man DESTINATION ${CMAKE_INSTALL_MANDIR}/man1 RENAME kodak-i2000-fw.1)
